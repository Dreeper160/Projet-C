medium/12_desktop_size.c

Ce programme montre comment il est possible de récupérer la taille de l'environnment de bureau de l'ordinateur.

#include <MLV/MLV_all.h>
//
// Attention !
// Pour pouvoir compiler ce programme sous windows et sous macintosh,
// il faut, pour la déclaration du main, respecter strictement la syntaxe
// suivante :
//
int main( int argc, char *argv[] ){
int width = MLV_get_desktop_width();
int height = MLV_get_desktop_height();
MLV_create_window(
"medium - 12 - Le résolution du bureau d'environement",
"résolution du bureau", width, height
);
MLV_draw_text(
20, 20,
"La résolution du bureau est de : %d X %d",
MLV_COLOR_GREEN,
width, height
);
MLV_actualise_window();
MLV_wait_seconds( 10 );
MLV_free_window();
return 0;
}
